# Experiment 7 - Three-address-code generation

**Manual experiment number:** 7

## Program and test inputs

### `input.txt`

```text

```

### `tac.l`

```lex
%option noyywrap nodefault
%{
#include "tac.tab.h"
#include <stdlib.h>
#include <string.h>

static char *copy_text(const char *text) {
    size_t length = strlen(text) + 1;
    char *copy = (char *)malloc(length);
    if (copy == NULL) {
        fprintf(stderr, "Out of memory\n");
        exit(EXIT_FAILURE);
    }
    memcpy(copy, text, length);
    return copy;
}
%}

%%
[a-zA-Z_][a-zA-Z0-9_]*   { yylval.text = copy_text(yytext); return ID; }
([0-9]+\.[0-9]+)|[0-9]+  { yylval.text = copy_text(yytext); return NUM; }
[ \t\r\n]+                { /* skip whitespace */ }
.                         { return yytext[0]; }
%%
```

### `tac.y`

```yacc
%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int yylex(void);
int yyerror(const char *message);

static int temporary_count = 1;

static char *copy_text(const char *text) {
    size_t length = strlen(text) + 1;
    char *copy = (char *)malloc(length);
    if (copy == NULL) {
        fprintf(stderr, "Out of memory\n");
        exit(EXIT_FAILURE);
    }
    memcpy(copy, text, length);
    return copy;
}

static char *new_temporary(void) {
    char buffer[32];
    snprintf(buffer, sizeof buffer, "t%d", temporary_count++);
    return copy_text(buffer);
}

static char *emit_binary(char *left, char operator, char *right) {
    char *temporary = new_temporary();
    printf("%s = %s %c %s\n", temporary, left, operator, right);
    free(left);
    free(right);
    return temporary;
}
%}

%define parse.error verbose
%union {
    char *text;
}
%token <text> ID NUM
%type <text> expression
%left '+' '-'
%left '*' '/'
%start statement

%%
statement:
      ID '=' expression ';' {
          printf("%s = %s\n", $1, $3);
          free($1);
          free($3);
      }
    ;

expression:
      ID                         { $$ = $1; }
    | NUM                        { $$ = $1; }
    | '(' expression ')'         { $$ = $2; }
    | expression '+' expression { $$ = emit_binary($1, '+', $3); }
    | expression '-' expression { $$ = emit_binary($1, '-', $3); }
    | expression '*' expression { $$ = emit_binary($1, '*', $3); }
    | expression '/' expression { $$ = emit_binary($1, '/', $3); }
    ;
%%

int main(void) {
    printf("Enter the expression:\n");
    return yyparse() == 0 ? EXIT_SUCCESS : EXIT_FAILURE;
}

int yyerror(const char *message) {
    fprintf(stderr, "Error: %s\n", message);
    return 0;
}
```

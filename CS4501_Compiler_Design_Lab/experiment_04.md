### `valvar.y`

```yacc
%{
#include <stdio.h>
#include <stdlib.h>

int yylex(void);
int yyerror(const char *message);
%}

%define parse.error verbose
%token LET DIG
%start input

%%
input:
      variable optional_newline
    ;

optional_newline:
      %empty
    | '\n'
    ;

variable:
      LET suffix
    ;

suffix:
      %empty
    | suffix LET
    | suffix DIG
    ;
%%

int main(void) {
    int status;
    printf("Enter the variable:\n");
    status = yyparse();
    if (status == 0) {
        printf("Valid variable\n");
        return EXIT_SUCCESS;
    }
    return EXIT_FAILURE;
}

int yyerror(const char *message) {
    printf("Invalid variable\n");
    fprintf(stderr, "Parser detail: %s\n", message);
    return 0;
}
```

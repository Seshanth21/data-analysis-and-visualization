# Experiment 1 - Lexical analyzer with symbol table

**Manual experiment number:** 1

## Program and test inputs

### `input.c`

```c
int a = 10; // sum variable
b = a + 5;
/* update the result */
float total = b * 2.5;
```

### `symtab.l`

```lex
%option noyywrap nodefault yylineno
%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_SYMBOLS 100
#define MAX_NAME_LEN 63

struct symbol {
    char name[MAX_NAME_LEN + 1];
};

static struct symbol symbols[MAX_SYMBOLS];
static int symbol_count = 0;

static int lookup(const char *name) {
    int i;
    for (i = 0; i < symbol_count; ++i) {
        if (strcmp(symbols[i].name, name) == 0) {
            return i;
        }
    }
    return -1;
}

static void insert_symbol(const char *name) {
    if (lookup(name) >= 0) {
        return;
    }
    if (symbol_count >= MAX_SYMBOLS) {
        fprintf(stderr, "Symbol table is full; cannot insert %s\n", name);
        exit(EXIT_FAILURE);
    }
    snprintf(symbols[symbol_count].name,
             sizeof symbols[symbol_count].name,
             "%s",
             name);
    ++symbol_count;
}
%}

DIGIT          [0-9]
ID             [a-zA-Z_][a-zA-Z0-9_]*
BLOCK_COMMENT  [/][*]([^*]|[*]+[^*/])*[*]+[/]

%%
{BLOCK_COMMENT}                         { printf("Comment : %s\n", yytext); }
"//"[^\n]*                              { printf("Comment : %s\n", yytext); }
{ID}                                    { insert_symbol(yytext); printf("Identifier : %s\n", yytext); }
{DIGIT}+(\.{DIGIT}+)?                   { printf("Constant : %s\n", yytext); }
"=="|"!="|"<="|">="|"++"|"--"|"&&"|"||" { printf("Operator : %s\n", yytext); }
[+\-*/=<>]                              { printf("Operator : %s\n", yytext); }
[ \t\r\n]+                              { /* skip whitespace */ }
.                                       { /* delimiters and other characters are ignored */ }
%%

int main(int argc, char **argv) {
    int i;
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <input-file>\n", argv[0]);
        return EXIT_FAILURE;
    }

    yyin = fopen(argv[1], "r");
    if (yyin == NULL) {
        perror(argv[1]);
        return EXIT_FAILURE;
    }

    yylex();
    fclose(yyin);

    printf("\nSYMBOL TABLE\n");
    printf("S.No\tName\n");
    for (i = 0; i < symbol_count; ++i) {
        printf("%d\t%s\n", i + 1, symbols[i].name);
    }
    return EXIT_SUCCESS;
}
```

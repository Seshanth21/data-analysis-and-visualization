### `control.y`

```yacc
%{
#include <stdio.h>
#include <stdlib.h>

int yylex(void);
int yyerror(const char *message);
%}

%define parse.error verbose
%token IF ELSE FOR WHILE SWITCH CASE DEFAULT BREAK
%token ID NUM EQ NE LE GE INC DEC
%nonassoc LOWER_THAN_ELSE
%nonassoc ELSE
%left '+' '-'
%left '*' '/'
%start program

%%
program:
      statement
    ;

statement:
      simple_statement
    | block
    | IF '(' condition ')' statement %prec LOWER_THAN_ELSE
    | IF '(' condition ')' statement ELSE statement
    | WHILE '(' condition ')' statement
    | FOR '(' for_init ';' for_condition ';' for_update ')' statement
    | switch_statement
    ;

block:
      '{' statement_list '}'
    ;

statement_list:
      %empty
    | statement_list statement
    ;

simple_statement:
      assignment ';'
    | expression ';'
    | BREAK ';'
    | ';'
    ;

assignment:
      ID '=' expression
    ;

condition:
      expression
    | expression relational_operator expression
    ;

relational_operator:
      EQ
    | NE
    | LE
    | GE
    | '<'
    | '>'
    ;

for_init:
      %empty
    | assignment
    ;

for_condition:
      %empty
    | condition
    ;

for_update:
      %empty
    | assignment
    | ID INC
    | ID DEC
    ;

switch_statement:
      SWITCH '(' expression ')' '{' case_section_list default_section_opt '}'
    ;

case_section_list:
      %empty
    | case_section_list case_section
    ;

case_section:
      CASE NUM ':' statement_list
    ;

default_section_opt:
      %empty
    | DEFAULT ':' statement_list
    ;

expression:
      ID
    | NUM
    | '(' expression ')'
    | expression '+' expression
    | expression '-' expression
    | expression '*' expression
    | expression '/' expression
    ;
%%

int main(void) {
    int status;
    printf("Enter a C control structure syntax:\n");
    status = yyparse();
    if (status == 0) {
        printf("Valid control structure syntax.\n");
        return EXIT_SUCCESS;
    }
    return EXIT_FAILURE;
}

int yyerror(const char *message) {
    printf("Invalid control structure syntax.\n");
    fprintf(stderr, "Parser detail: %s\n", message);
    return 0;
}
```

# Experiment 10 - Compiler back end for 8086-style assembly

**Manual experiment number:** 10

## Program and test inputs

### `backend.l`

```lex
%option noyywrap nodefault
%{
#include "backend.tab.h"
#include <stdlib.h>
#include <string.h>

static char *copy_text(const char *text) {
    size_t length = strlen(text) + 1;
    char *copy = (char *)malloc(length);
    if (copy == NULL) {
        fprintf(stderr, "Out of memory\n");
        exit(EXIT_FAILURE);
    }
    memcpy(copy, text, length);
    return copy;
}
%}

%%
[a-zA-Z_][a-zA-Z0-9_]*   { yylval.text = copy_text(yytext); return ID; }
[0-9]+                    { yylval.text = copy_text(yytext); return NUM; }
[=+\-*/;]                  { return yytext[0]; }
[ \t\r\n]+                { /* skip whitespace */ }
.                         { return yytext[0]; }
%%
```

### `backend.y`

```yacc
%{
#include <stdio.h>
#include <stdlib.h>

int yylex(void);
int yyerror(const char *message);

static void emit_assignment(const char *destination,
                            const char *left,
                            char operator,
                            const char *right) {
    printf("MOV AX, %s\n", left);
    switch (operator) {
        case '+':
            printf("ADD AX, %s\n", right);
            break;
        case '-':
            printf("SUB AX, %s\n", right);
            break;
        case '*':
            printf("MUL %s\n", right);
            break;
        case '/':
            printf("MOV DX, 0\n");
            printf("MOV BX, %s\n", right);
            printf("DIV BX\n");
            break;
        default:
            break;
    }
    printf("MOV %s, AX\n\n", destination);
}
%}

%define parse.error verbose
%union {
    char *text;
}
%token <text> ID NUM
%type <text> operand
%start statement_list

%%
statement_list:
      %empty
    | statement_list statement
    ;

statement:
      ID '=' operand ';' {
          printf("MOV AX, %s\n", $3);
          printf("MOV %s, AX\n\n", $1);
          free($1);
          free($3);
      }
    | ID '=' operand '+' operand ';' {
          emit_assignment($1, $3, '+', $5);
          free($1); free($3); free($5);
      }
    | ID '=' operand '-' operand ';' {
          emit_assignment($1, $3, '-', $5);
          free($1); free($3); free($5);
      }
    | ID '=' operand '*' operand ';' {
          emit_assignment($1, $3, '*', $5);
          free($1); free($3); free($5);
      }
    | ID '=' operand '/' operand ';' {
          emit_assignment($1, $3, '/', $5);
          free($1); free($3); free($5);
      }
    ;

operand:
      ID  { $$ = $1; }
    | NUM { $$ = $1; }
    ;
%%

int main(void) {
    printf("Enter TAC statements (end with Ctrl+D):\n");
    return yyparse() == 0 ? EXIT_SUCCESS : EXIT_FAILURE;
}

int yyerror(const char *message) {
    fprintf(stderr, "Syntax Error: %s\n", message);
    return 0;
}
```

### `art_expr.y`

```yacc
%{
#include <stdio.h>
#include <stdlib.h>

int yylex(void);
int yyerror(const char *message);
%}

%define parse.error verbose
%token ID NUM
%left '+' '-'
%left '*' '/'
%right UMINUS
%start input

%%
input:
      expression optional_newline
    ;

optional_newline:
      %empty
    | '\n'
    ;

expression:
      expression '+' expression
    | expression '-' expression
    | expression '*' expression
    | expression '/' expression
    | '-' expression %prec UMINUS
    | '(' expression ')'
    | NUM
    | ID
    ;
%%

int main(void) {
    int status;
    printf("Enter the Expression\n");
    status = yyparse();
    if (status == 0) {
        printf("Valid Expression\n");
        return EXIT_SUCCESS;
    }
    return EXIT_FAILURE;
}

int yyerror(const char *message) {
    printf("Invalid Expression\n");
    fprintf(stderr, "Parser detail: %s\n", message);
    return 0;
}
```
